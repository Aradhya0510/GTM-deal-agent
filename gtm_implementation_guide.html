UC Functions — agent tools for deal intelligence

Python02_uc_tools.py
from databricks.sdk import WorkspaceClient
from databricks.sdk.service.catalog import FunctionParameterType

w = WorkspaceClient()

# Tool 1: Calculate live deal health score
w.functions.create(
    name="gtm.tools.calculate_deal_health",
    catalog_name="gtm",
    schema_name="tools",
    input_params=[{"name": "opp_id", "type_text": "STRING", "position": 0}],
    return_type_text="STRING",
    routine_body="EXTERNAL",
    routine_definition="""
      SELECT json_object(
        'health_score',  ROUND(
            (CASE WHEN stage IN ('Proposal','Negotiation') THEN 30 ELSE 10 END) +
            (CASE WHEN last_contact_days < 7 THEN 20 ELSE 0 END) +
            (CASE WHEN num_stakeholders >= 3 THEN 25 ELSE 10 END) +
            (CASE WHEN close_date > CURRENT_DATE THEN 15 ELSE 0 END)
        , 1),
        'stage',         stage,
        'days_to_close', DATEDIFF(close_date, CURRENT_DATE),
        'risk_flags',    risk_flags,
        'champion',      champion_name
      ) FROM gtm.opportunities_enriched
      WHERE opp_id = :opp_id
    """,
    comment="Returns deal health score (0-100) and risk flags for an opportunity"
)

# Tool 2: Get account signals (engagement, product usage, sentiment)
w.functions.create(
    name="gtm.tools.get_account_signals",
    catalog_name="gtm",
    schema_name="tools",
    input_params=[{"name": "account_id", "type_text": "STRING", "position": 0}],
    return_type_text="STRING",
    routine_body="EXTERNAL",
    routine_definition="""
      SELECT json_object(
        'product_dau',       avg_dau_last_30d,
        'feature_adoption',  top_features_used,
        'support_tickets',   open_p1_tickets,

Python03_playground_export.py
import mlflow
from databricks.agents import ResponsesAgent
from databricks.agents.tools import VectorSearchRetriever, UCFunctionTool

mlflow.set_experiment("/gtm/deal-intelligence-agent")

# System prompt — ground the agent in GTM context
SYSTEM_PROMPT = """You are an expert GTM intelligence assistant for B2B SaaS account executives.

You have access to:
- Live CRM data and deal health calculations
- Gong call transcripts and account history
- Competitive battlecards and win/loss stories
- Product usage and engagement signals

When asked about a deal or account:
1. ALWAYS calculate the current deal health score first
2. Retrieve the 3 most relevant call transcripts for context
3. Check for any active risk flags before making recommendations
4. Ground all outreach in specific signals (usage data, conversation topics, stakeholder names)

Respond with concrete, actionable intelligence — not generic advice."""

# Build agent with tools
agent = ResponsesAgent(
    model="databricks/claude-3-5-sonnet",
    tools=[
        VectorSearchRetriever(
            index_name="gtm.vectors.call_transcripts",
            columns=["transcript_text", "call_date", "participants", "summary"],
            num_results=3
        ),
        VectorSearchRetriever(
            index_name="gtm.vectors.battlecards",
            columns=["competitor", "use_case", "win_themes", "objection_handlers"],
            num_results=2
        ),
        UCFunctionTool("gtm.tools.calculate_deal_health"),
        UCFunctionTool("gtm.tools.get_account_signals"),
    ],
    system_prompt=SYSTEM_PROMPT
)

# Test query
result = agent.predict({
    "messages": [{
        "role": "user",
        "content": "What's the deal health on opportunity OPP-4821 (Acme Corp) and draft a follow-up email for tomorrow's QBR?"
    }]
})
print(result["messages"][-1]["content"])

from databricks.agent_bricks import AgentBricks

# 1. Task-specific LLM judges (personalization, groundedness, tone)

optimizer = AgentBricks.configure(
    task_description="""
        Given an account's CRM data, Gong transcript history, product usage signals,
        and a specific opportunity context, generate a personalized outreach email
        for the AE to review and send. The email should:
        - Reference a specific insight from the last Gong call by name
        - Connect product usage patterns to a business outcome the champion cares about

5 Govern & secure

Apply Unity Catalog row-level security so reps only see their territory, mask PII fields, and configure Lakewatch to detect prompt injection and data exfiltration.

Unity Catalog — territory-based row security

SQL08_uc_governance.sql
-- Row-level security: AEs only see their territory's accounts
CREATE ROW FILTER gtm.security.territory_filter
  ON gtm.opportunities (territory STRING)
  AS CURRENT_USER() = ae_owner
  OR IS_MEMBER('sales-leadership')
  OR IS_MEMBER('revenue-operations');

ALTER TABLE gtm.opportunities
  ADD ROW FILTER gtm.security.territory_filter ON (territory);

-- Column masking: hide personal email for non-managers
CREATE COLUMN MASK gtm.security.mask_personal_email
  ON gtm.contacts (personal_email STRING)
  AS CASE
    WHEN IS_MEMBER('sales-leadership') THEN personal_email
    ELSE CONCAT(LEFT(personal_email, 2), '***@***.***')
  END;

ALTER TABLE gtm.contacts
  ADD COLUMN MASK gtm.security.mask_personal_email ON (personal_email);

-- Audit: capture every agent read of opportunity data
CREATE TABLE gtm.audit.agent_data_access (
  trace_id      VARCHAR,
  ae_id         VARCHAR,
  opp_id        VARCHAR,
  tool_called   VARCHAR,
  accessed_at   TIMESTAMP DEFAULT NOW()
);

Lakewatch — agent security monitoring

Python09_lakewatch.py
from databricks.lakewatch import DetectionRule, Severity

# Rule 1: Detect contact data exfiltration via agent output
DetectionRule.create(
    name="gtm_contact_pii_in_output",
    description="Alert when agent output contains PII patterns beyond the requesting AE's accounts",
    severity=Severity.HIGH,
    query="""
        SELECT trace_id, ae_id, output_text, account_ids_accessed
        FROM gtm.mlflow_traces
        WHERE trace_timestamp > CURRENT_TIMESTAMP - INTERVAL 1 HOUR
          AND ARRAY_SIZE(account_ids_accessed) > 20  -- Unusual breadth
          AND output_text RLIKE '\\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\\.[A-Z]{2,}\\b'
    """,
    alert_channel="slack://security-alerts"
)

# Rule 2: Detect prompt injection attempts in user inputs
DetectionRule.create(
    name="gtm_prompt_injection",
    description="Detect prompt injection patterns in AE inputs to the agent",
    severity=Severity.CRITICAL,
    query="""
        SELECT trace_id, ae_id, user_input
        FROM gtm.mlflow_traces
        WHERE trace_timestamp > CURRENT_TIMESTAMP - INTERVAL 15 MINUTES
          AND (
            LOWER(user_input) LIKE '%ignore previous instructions%'
            OR LOWER(user_input) LIKE '%system prompt%'
            OR LOWER(user_input) LIKE '%reveal your instructions%'
            OR LOWER(user_input) LIKE '%act as%'
          )
    """,
    alert_channel="pagerduty://security-oncall",
    auto_block=True  # Suspend the trace immediately
)

# Rule 3: Unusual outreach volume spike (potential abuse)
DetectionRule.create(
    name="gtm_outreach_volume_spike",
    description="Alert when a single AE generates 10x their normal daily outreach volume",
    severity=Severity.MEDIUM,
    query="""
        WITH daily_avg AS (
            SELECT ae_id, AVG(daily_count) AS avg_count
            FROM gtm.outreach_log_daily_agg
            WHERE log_date > CURRENT_DATE - 30
            GROUP BY ae_id
        )
        SELECT t.ae_id, t.today_count, d.avg_count
        FROM gtm.outreach_log_today t
        JOIN daily_avg d ON t.ae_id = d.ae_id
        WHERE t.today_count > d.avg_count * 10
    """,
    alert_channel="slack://revenue-ops"
)

from databricks.agents import deploy

w = WorkspaceClient()

# Deploy the evaluated agent to a REST endpoint
deployment = deploy(
    model_name="gtm.agents.deal-intelligence",
    version=mlflow.register_model(
        model_uri=f"runs:/{run_id}/gtm_agent",
        name="gtm.agents.deal-intelligence"
    ).version,
    scale_to_zero=True,           # Serverless GPU — cost-efficient off-hours
    enable_feedback_ui=True,      # AEs can rate responses → feeds Quotient RL loop
    traffic_config={
        "canary_percent": 10,     # 10% of AEs get v1.2 first
        "rollout_hours": 48       # Full rollout over 48h if no regressions
    }
)

print(f"Endpoint: {deployment.endpoint_url}")
print(f"Review UI: {deployment.review_app_url}")

# ── Quotient AI continuous improvement loop ───────────────────────────
# Automatically active post-deployment:
# 1. AE feedback (thumbs up/down on outreach drafts) flows to Quotient
# 2. Quotient monitors answer quality, detects regressions, pins failures
# 3. Reinforcement learning loop retrains on accepted/rejected outreach
# 4. Improved agent versions staged automatically for evaluation
# 5. Alerts fired to RevOps if quality drops > 5% week-over-week

# ── Databricks Apps (AE-facing UI) ───────────────────────────────────
app = w.apps.create(
    name="gtm-deal-intelligence",
    description="Deal health, competitive intel, and personalized outreach for AEs",
    resources=[{"name": "deal-intelligence-endpoint", "serving_endpoint": {"name": deployment.endpoint_name}}]
)
print(f"AE App URL: {app.url}")

Testing strategy

Unit tests — tool level

Assert calculate_deal_health() returns correct risk flags for known fixtures. Test Vector Search returns ≥1 result for 10 seed queries. Run in CI on every commit.

Integration tests — agent level

30 golden scenarios with AE-rated expected outputs. Assert personalization score ≥ 0.80 and groundedness ≥ 0.90 before any promotion to staging.

Shadow mode

Run v1.2 alongside v1.1 for 2 weeks on 20% of traffic without surfacing results to AEs. Compare win rates on deals where AEs used agent outreach vs manual.

Regression gate (Quotient AI)

Automatic weekly regression check. If personalization quality drops > 5% or groundedness < 0.85, block promotion and page RevOps.
